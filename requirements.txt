# --- mediapipe is deliberately NOT listed in this file ----------------------
# It is installed by the Dockerfile as a separate, standalone step:
#     pip install --no-deps mediapipe==0.10.33
# If mediapipe appears anywhere in this file -- pinned or not -- pip resolves
# its declared dependency on opencv-contrib-python (the GL-linked build), which
# collides with opencv-python-headless below and makes `import cv2` die with
# libGL.so.1 missing. --no-deps in a separate command never triggers that
# resolution.

# CPU-only torch, deliberately. Railway has no GPU, and plain torch==2.2.2 from
# PyPI pulls in several GB of unusable CUDA packages.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.2.2+cpu

# numpy pinned below 2.0 to match torch 2.2.x, which is built against the NumPy
# 1.x C ABI. To move to numpy 2.x, upgrade torch to >=2.4 first and re-verify
# /v1/restore.
numpy>=1.26,<2

opencv-python-headless>=4.9,<4.11

# mediapipe 0.10.33's own dependencies, minus opencv-contrib-python. Listed
# here because mediapipe is installed with --no-deps and will not pull them in.
absl-py~=2.3
certifi
sounddevice~=0.5
flatbuffers~=25.9
matplotlib>=3.8,<4

fastapi>=0.110
uvicorn[standard]>=0.29
python-multipart>=0.0.9
